\subsection{Dual Space and Dual Map}

\setcounter{thm}{107}
%\textbf{3.108}
\begin{mydef}
  A \qt{linear functional} $\phi$ is an element of $\mathcal{L}(V, \mathbb{F})$. So, $\phi \in \linmap(V, \myF)$.
\end{mydef}

\textbf{Examples:}
\begin{equation}
  \begin{array}{lll}
    \phi: \mathbb{R}^3  \to \mathbb{R}, &\phi (x,y,z)  & \mapsto 4x-5y-2z \\
    \phi: \mathbb{F}^n  \to \mathbb{F}, &\phi (x_1, \dots, x_n)
    & \mapsto c_1x_1 + \dots + c_nx_n  \\
    \phi: \mathcal{P} (\mathbb{R})  \to \mathbb{R},
    & \phi(p) & \mapsto 3p''(5) + 7p(4) \\
    \phi: \mathcal{P}(\mathbb{R}) \to \mathbb{R},
    & \phi(p)  &\mapsto \textstyle \int_{0}^{1} p(x) dx
  \end{array}
\end{equation}

\setcounter{thm}{109}
%\textbf{3.110}
\begin{thm}
  The dual space of $V$, denoted by $\dual{V}$ or $V^{*}$, is the vector space of all linear functionals on $V$, i.e.,
  \begin{equation}
    \dual{V} := \linmap(V, \myF).
  \end{equation}
\end{thm}

%\textbf{3.111}
\begin{thm} [$\dim \dual{V} = \dim V$]
  \label{thm: dim V' = dim V}
  Suppose $V$ is finite-dimensional. Then $\dual{V}$ is also finite-dimensional, and
  \begin{equation}
    \dim \dual{V} = \dim V.
  \end{equation}
\end{thm}
\begin{prf}
  By \ref{thm: dimension of L(V,W) = dim L (V) * dim W (W)}, we have
  $\dim \dual{V} = \dim \mathcal{L}(V, \myF)=(\dim V) \cdot (\dim \myF) = \dim V.$
\end{prf}

\begin{mydef}
  If $\onetillm{v}$ is a basis of $V$, then the \qt{dual basis} of $\onetillm{v}$ is the list $\varphi_1, \dots, \varphi_m \in \dual{V}$, where each $\varphi_j$ is the linear functional such that
  \begin{equation}
    \varphi_j(v_k) = \delta_{j,k} =
  \begin{cases}
    1,  & \text{if $k=j$} \\
    0, & \text{if $k \neq j$}.
  \end{cases}
  \end{equation}
  Note that this is not the definition of each $\varphi_j$ but a property of each $\varphi_j$.
\end{mydef}

\setcounter{thm}{113}
\begin{thm}
  Let $\onetillm{v}$ be a basis of $V$. $\onetillm{\varphi}$ is called \qt{a dual basis} of $\onetillm{v}$ because, for all $v \in V$:
  \begin{equation}
    v=\varphi_1 (v)\cdot v_1 + \cdots + \varphi_m(v)\cdot v_m.
  \end{equation}
\end{thm}
\begin{prf}
  Let $v \in V$ such that $v= c_1v_1 + \cdots + c_mv_m$. Applying
  $\varphi_j$ on both sides gives $\varphi_j(v)=c_j$ for $j = 1, \ddd, m$. Hence,
  $v=\varphi_1 (v)v_1 + \cdots + \varphi_m(v)v_m$.
\end{prf}

%\textbf{3.116}
\setcounter{thm}{115}
\begin{thm}[basis of the dual space]
  The dual basis of a basis of $V$ is a basis of the dual space $\dual{V}$.
\end{thm}
\begin{prf}
  Suppose $v_1, \ddd, v_n$ is a basis of $V$. Let $\varphi_1, \ddd, \varphi_n$ denote its dual basis. Suppose
  \[
    a_1 \varphi_1 + \cdots + a_n \varphi_n = 0 \text{ for some } a_1, \ddd, a_n \in \myF.
  \]

  Which means that we get a function in $\linmap(V, \myF)$ that always returns $0$. In other words, $(a_1 \varphi_1 + \cdots + a_n \varphi_n)(v)=0$ for all $v \in V$. This implies
  \[
    (a_1 \varphi_1 + \cdots + a_n \varphi_n)(v_k)=a_k \quad \forall k \in \{1, \ddd, n\}.
  \]

  Thus, $a_1 = \cdots = a_n = 0$. Hence, the list $\varphi_1, \ddd, \varphi_n$ is linearly independent. Since the list is a linearly independent list in $\dual{V}$ whose length equals $\dim \dual{V}$ by \ref{thm: dim V' = dim V}, we conclude that $\varphi_1, \ddd, \varphi_n$ is a basis of $\dual{V}$ by \ref{thm: linearly independent list of the right length is a basis}.
\end{prf}

\setcounter{thm}{117}
\begin{mydef}[dual map, $\dual{T}$]
  The \qt{dual map} $\dual{T}$ of $T \in \lin{V}{W}$ is the linear map $\dual{T} \in \lin{W'}{V'} = \lin{\lin{W}{\myF}}{\lin{V}{\myF}}$ defined as follows:
\begin{equation}
    \dual{T}(\varphi) := \varphi \circ T \quad \text{for every } \varphi \in \dual{W}=\lin{W}{\myF}.
\end{equation}

    Therefore, we have that $\dual{T}(\varphi) \in \dual{V} = \lin{V}{\myF}$. Hence, $\dual{T} \in \lin{\dual{W}}{\dual{V}}$ is indeed a linear map from $\dual{W}$ to $\dual{V}$, since:
\begin{itemize}
      \item $\varphi, \psi \in \dual{W} \implies T' (\varphi + \psi) = (\varphi + \psi) \circ T = \varphi \circ T + \psi \circ T = T' (\varphi) + T'(\psi)$
    \item $\lambda \in \mathbb{F}, \varphi \in W' \implies T' (\lambda \varphi) = (\lambda \varphi) \circ T = \lambda (\varphi \circ T) = \lambda T' (\varphi)$.
\end{itemize}


\end{mydef}

\setcounter{thm}{119}
\begin{thm} [algebraic properties of dual maps]
  \label{algebraic-properties-of-dual-maps}
  $T \in \linmap(V,W) \implies$
  \begin{enumerate}[label=\textbf{(\alph*)}]
    \item $\dual{(S+T)} = \dual{S} + \dual{T} \quad \forall S \in \linmap(V,W)$
    \item $\dual{(\lambda T)} = \lambda \dual{T} \quad \quad \forall \lambda \in \myF$
    \item $\dual{(ST)} = \dual{T}\dual{S} \quad \; \forall S \in \linmap (W,U)$
  \end{enumerate}
\end{thm}



\subsection{Dual Space and Range of Dual of Linear Map}

\begin{mydef} [annihilator, $U^0$]
  \label{def: annihilator}
  For $U \subseteq V$, the \qt{annihilator} of $U$, denoted by $U^{0}$, is defined by
  \begin{equation}
    U^0 = \{ \varphi \in \dual{V} \; \mid \; \varphi (u) = 0 \quad \forall u \in U \}.
  \end{equation}
\end{mydef}

% 3.124
\setcounter{thm}{123}
\begin{thm}[the annihilator is a subspace]
  \label{thm: the annihilator is a subspace}
  If $U \subseteq V$, then $U^{0}$ is a subspace of $\dual{V}$.
\end{thm}

% 3.125
\setcounter{thm}{124}
\begin{thm} [dimension of the annihilator]
  \label{thm: dimension of the annihilator}
  If $U\subseteq V$ and $\dim V < \infty$, then
  \begin{equation}
    \dim U^0 = \dim V - \dim U.
  \end{equation}
\end{thm}
\begin{prf}
  Let $i \in \linmap(U, V)$ be the inclusion map defined by $i(u) = u$ for all $u \in U$.
  Define the dual map $i' \in \linmap(\dual{V}, \dual{U})$ by $i'(\varphi) = \varphi \circ i = \left.\varphi\right|_U$.
  Then $\mynull i' = \{\varphi \in \dual{V} \mid \left.\varphi\right|_U = 0\} = U^0$.
  Furthermore, by extending a basis of $U$ to a basis of $V$, every linear functional on $U$ extends to a linear functional on $V$, so $i'$ is surjective ($\myrange i' = \dual{U}$).
  By the Rank-Nullity Theorem (\ref{rank-nullity-theorem}) applied to $i'$:
  \begin{equation}
    \dim \dual{V} = \dim \mynull i' + \dim \myrange i' \implies \dim V = \dim U^0 + \dim U.
  \end{equation}
  Thus, $\dim U^0 = \dim V - \dim U$.
\end{prf}

% 3.127
\setcounter{thm}{126}
\begin{thm} [condition for the annihilator to equal $\{0\}$ or the whole space]
  If $\dim V < \infty$ and $U\subseteq V$, then:
  \begin{enumerate}[label=\textbf{(\alph*)}]
    \item $U^0 = \{0 \} \iff U = V$
    \item $U^0 = \dual{V} \iff U = \{0\}$
   \end{enumerate}
\end{thm}
\begin{prf}
  Follows directly from \ref{thm: dimension of the annihilator}:
  \begin{enumerate}[label=\textbf{(\alph*)}]
    \item $U^0 = \{0\} \iff \dim U^0 = 0 \iff \dim V - \dim U = 0 \iff \dim U = \dim V \iff U = V$.
    \item $U^0 = \dual{V} \iff \dim U^0 = \dim V \iff \dim V - \dim U = \dim V \iff \dim U = 0 \iff U = \{0\}$.
  \end{enumerate}
  \vspace{-\baselineskip}
\end{prf}

% 3.128
\begin{thm}[the null space of $\dual{T}$]
  \label{thm: the null space of dual T}
  If $\dim V < \infty$, $\dim W<\infty$, and $T \in \linmap (V,W)$, then:
  \begin{enumerate}[label=\textbf{(\alph*)}]
    \item $\mynull \dual{T} = (\myrange T)^0$
    \item $\dim \mynull \dual{T} = \dim \mynull T + \dim W - \dim V$
  \end{enumerate}
\end{thm}
\begin{prf}
  \begin{enumerate}[label=\textbf{(\alph*)}]
    \item Let $\varphi \in \dual{W}$. Then $\varphi \in \mynull \dual{T} \iff \dual{T}(\varphi) = 0 \iff \varphi \circ T = 0 \iff \varphi(Tv) = 0 \quad \forall v \in V \iff \varphi \in (\myrange T)^0$.
    \item Using part (a) and \ref{thm: dimension of the annihilator}:
    \begin{equation}
      \dim \mynull \dual{T} = \dim (\myrange T)^0 = \dim W - \dim \myrange T.
    \end{equation}
    Substituting $\dim \myrange T = \dim V - \dim \mynull T$ (by Rank-Nullity) gives $\dim \mynull \dual{T} = \dim \mynull T + \dim W - \dim V$.
  \end{enumerate}
  \vspace{-\baselineskip}
\end{prf}

% 3.129
\begin{thm}[$T$ surjective $\iff \dual{T}$ injective]
  \label{thm: T surjective iff dual T injective}
  If $\dim V < \infty$, $\dim W<\infty$, and $T \in \linmap (V,W)$, then:
  \begin{equation}
    T \text{ is surjective} \iff \dual{T} \text{ is injective.}
  \end{equation}
\end{thm}
\begin{prf}
  $T$ is surjective $\iff \myrange T = W \iff (\myrange T)^0 = \{0\} \iff \mynull \dual{T} = \{0\}$ (by \ref{thm: the null space of dual T}(a)) $\iff \dual{T}$ is injective.
\end{prf}

% 3.130
\begin{thm}[the range of $\dual{T}$]
  \label{thm: the range of dual T}
  If $\dim V < \infty$, $\dim W<\infty$, and $T \in \linmap (V,W)$, then:
  \begin{enumerate}[label=\textbf{(\alph*)}]
    \item $\dim \myrange \dual{T} = \dim \myrange T$
    \item $\myrange \dual{T} = (\mynull T)^0$
  \end{enumerate}
\end{thm}
\begin{prf}
  \begin{enumerate}[label=\textbf{(\alph*)}]
    \item By Rank-Nullity on $\dual{T} \in \linmap(\dual{W}, \dual{V})$ and \ref{thm: the null space of dual T}(a):
    \begin{equation}
      \dim \myrange \dual{T} = \dim \dual{W} - \dim \mynull \dual{T} = \dim W - \dim(\myrange T)^0 = \dim \myrange T.
    \end{equation}
    \item If $\psi \in \myrange \dual{T}$, then $\psi = \dual{T}(\varphi) = \varphi \circ T$ for some $\varphi \in \dual{W}$. For any $u \in \mynull T$, $\psi(u) = \varphi(Tu) = \varphi(0) = 0$, which shows $\psi \in (\mynull T)^0$. Thus $\myrange \dual{T} \subseteq (\mynull T)^0$.
    Since $\dim \myrange \dual{T} = \dim \myrange T = \dim V - \dim \mynull T = \dim (\mynull T)^0$, the two subspaces must be equal: $\myrange \dual{T} = (\mynull T)^0$.
  \end{enumerate}
  \vspace{-\baselineskip}
\end{prf}

% 3.131
\begin{thm} [$T$ injective $\iff \dual{T}$ surjective]
  If $\dim V < \infty$, $\dim W<\infty$, and $T \in \linmap (V,W)$, then:
  \begin{equation}
    T \text{ is injective} \iff \dual{T} \text{ is surjective.}
  \end{equation}
\end{thm}
\begin{prf}
  $T$ is injective $\iff \mynull T = \{0\} \iff (\mynull T)^0 = \dual{V} \iff \myrange \dual{T} = \dual{V}$ (by \ref{thm: the range of dual T}(b)) $\iff \dual{T}$ is surjective.
\end{prf}

\subsection{Matrix of Dual of Linear Map}

% 3.132
\setcounter{thm}{131}
\begin{thm}[matrix of $\dual{T}$ is the transpose of matrix of $T$]
  Suppose $V, W$ are finite-dimensional and $T \in \linmap(V,W)$. Let $v_1, \dots, v_n$ be a basis of $V$ and $w_1, \dots, w_m$ be a basis of $W$, with corresponding dual bases $\varphi_1, \dots, \varphi_n$ and $\psi_1, \dots, \psi_m$. Then
  \begin{equation}
    \mmatrix(\dual{T}) = (\mmatrix(T))^{\top}.
  \end{equation}
\end{thm}
\begin{prf}
  Let $A = \mmatrix(T)$ and $B = \mmatrix(\dual{T})$.
  For $k \in \{1, \dots, n\}$ and $j \in \{1, \dots, m\}$, we have $Tv_k = \sum_{r=1}^m A_{r,k} w_r$.
  Applying the definition of the dual map gives
  \begin{equation}
    (\dual{T}\psi_j)(v_k) = \psi_j(Tv_k) = \psi_j\left(\sum_{r=1}^m A_{r,k} w_r\right) = \sum_{r=1}^m A_{r,k} \psi_j(w_r) = A_{j,k}.
  \end{equation}
  On the other hand, $\dual{T}\psi_j = \sum_{r=1}^n B_{r,j} \varphi_r$, so evaluating at $v_k$ gives $(\dual{T}\psi_j)(v_k) = B_{k,j}$.
  Thus, $B_{k,j} = A_{j,k}$, which means $\mmatrix(\dual{T}) = (\mmatrix(T))^\top$.
\end{prf}

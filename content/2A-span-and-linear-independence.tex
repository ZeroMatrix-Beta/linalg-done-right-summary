\section{Span and Linear Independence}
\subsection{Linear Combinations and Span}

\setcounter{thm}{1}
\begin{mydef} [linear combination]
  $a_1 v_1 + \cdots + a_m v_m$ is called a \qt{linear combination} of a list of vectors $v_1, \dots, v_m$ ($a_1, \ddd, a_m \in \myF$).
\end{mydef}

\setcounter{thm}{3}
\begin{mydef} [span of a list of vectors]
	The \qt{span} of a list of vectors $\onetillm{v}$ is defined as follows:
  \begin{equation}
    \begin{aligned}
      \myspan {v_1, \dots, v_m} &:\equiv \{a_1 v_1 + \cdots + a_m v_m \mid a_1, \dots, a_m \in \mathbb{F} \}\\
      \myspan{\; } &:\equiv \{ 0 \} \mytext { (The span of the empty list $(\;)$ is defined to be $\{0\}$)}
    \end{aligned}
  \end{equation}

  The definition also implies that $\myspan{ 0} = \{0\}$, or written differently, $\myspan{\vec 0} = \{\vec 0\}$. Note that sometimes we write $\vec 0$ and sometimes we write $0$ throughout this document. Later on, $0$ sometimes also means the zero-operator.
\end{mydef}

\mce{6}
\begin{thm} [the span of a list of vectors is a subspace]
  \label{thm: the span of a list of vectors is a subspace}
  The span of a list of vectors in $V$ is the smallest subspace of $V$ containing all vectors in the list.
\end{thm}
\begin{prf}
  Let $v_1, \ddd, v_m \in V$. We have
  \begin{equation}
    0 = 0v_1 + \cdots + 0v_m \in \myspan{v_1, \ddd, v_m},
  \end{equation}

  and hence, the additive identity is in $\myspan{v_1, \ddd, v_m}$.

  $\myspan{v_1, \ddd, v_m}$ is closed under addition, because $\forall$ $a$'s and $c$'s in $\myF$, we have
  \[
    \underbrace{(a_1 v_1 + \cdots + a_m v_m)}_{\in \; \myspan{v_1, \ddd, v_m}} + \underbrace{(c_1 v_1 + \cdots + c_m v_m)}_{\in \; \myspan{v_1, \ddd, v_m}} = (a_1 + c_1)v_1 + \cdots + (a_m + c_m)v_m \in \myspan{v_1, \ddd, v_m}.
  \]

  Furthermore, $\myspan{v_1, \ddd, v_m}$ is closed under scalar multiplication because for $\lambda \in \myF$, we have:
  \[
    \lambda \underbrace{(a_1 v_1 + \cdots + a_m v_m)}_{\in \; \myspan{v_1, \ddd, v_m}} = \lambda a_1 v_1 + \cdots + \lambda a_m v_m \in \myspan{v_1, \ddd, v_m}.
  \]

  Thus, $\myspan{v_1, \ddd, v_m}$ is a subspace of $V$ by \ref{thm: conditions for a subspace}.
  Every subspace of $V$ that contains each $v_k$ is closed under scalar multiplication and addition, and therefore, contains $\myspan{v_1, \ddd, v_m}$. Hence, $\myspan{v_1, \ddd, v_m}$ is the smallest subspace containing each $v_1, \ddd, v_m$. The fact that
  \begin{equation}
    v_j \in \myspan{v_1, \ddd, v_m} \quad (j = 1, \ddd, m)
  \end{equation}

  is obvious.
\end{prf}


% 2.8
\begin{mydef} [spanning a vector space]
  If $V=\myspan{\oneTillM{v}}$, we say that the list $\oneTillM{v}$ \qt{spans} $V$.
\end{mydef}

% 2.9
\setcounter{thm}{8}
\begin{mydef} [finite-dimensional vector space]
  A vector space $V$ is called \qt{finite-dimensional} if some list of vectors in $V$ spans $V$. For such a vector space, we write
  \begin{equation}
    \dim V \neq \infty \quad \text{or} \quad \dim V < \infty.
  \end{equation}
\end{mydef}


% 2.10
\begin{mydef}[polynomial, $\polyn(\myF)$]
  A function $p: \myF \to \myF$ is called a \qt{polynomial} with coefficients in $\myF$ if there exist $a_0, \ldots, a_m \in \myF$ \st
  \begin{equation}
    p(z)=a_0 + a_1z + a_2z^2 + \cdots + a_m z^m \quad \forall z \in \myF.
  \end{equation}

  We further denote
  \begin{itemize}
    \item $\polyn (\myF)$ as the set of all polynomials with coefficients in $\myF$.
    \item Note that $\polyn (\myF)$ is a vector space over $\myF$ and a subspace of $\myF^\myF$.
  \end{itemize}
\end{mydef}

% 2.11
\begin{mydef}[degree of a polynomial, $\deg p$]
  The degree $m$ of a non-zero polynomial $p(z)=a_0+a_1z+a_2z^2+\cdots+a_mz^m$ (with $a_m \neq 0$) is denoted by
  \begin{equation}
    \deg p := m.
  \end{equation}

  We also define for the zero polynomial $p=0$:
  \begin{equation}
    \deg 0 := -\infty.
  \end{equation}
\end{mydef}

% 2.12
\begin{mydef} [the set of all polynomials of degree at most $m$]
  For $m \ge 0$, $\polyn_m (\myF )=\myspan{1,z,\dots,z^m}$ denotes the set of all polynomials with coefficients in $\myF$ and degree at most $m$. One should actually use the lambda notation $\lambda z.z^m$ to speak of functions and write $\polyn_m (\myF )=\myspan{\lambda z.1,\lambda z.z,\dots,\lambda z.z^m}$, but this is not done most of the time. Note that $\dim \polyn_m(\myF) = m+1$.
\end{mydef}

\subsection{Linear Independence}

\setcounter{thm}{14}
\begin{mydef} [linear independence]
  A list of vectors $\oneTillM{v}$ in $V$ is called \qt{\lid} if the only choice of scalars $a_1, \dots, a_m \in \myF$ satisfying
  \begin{equation}
    a_1 v_1 + \dots + a_m v_m = 0
  \end{equation}
  is $a_1 = \cdots = a_m = 0$.
  is $a_1 = \cdots = a_m = 0$. In other words, the list is called \lid{} if the only way to combine the vectors together and yield zero is if we choose every coefficient to be zero.
  Equivalently, a list of vectors $\oneTillM{v}$ is linearly independent if and only if every vector $w \in \myspan{\oneTillM{v}}$ has a unique representation as a linear combination of $\oneTillM{v}$.
  Indeed, if $w = a_1 v_1 + \dots + a_m v_m = c_1 v_1 + \dots + c_m v_m$, subtracting gives
  \begin{equation}
    (a_1 - c_1) v_1 + \dots + (a_m - c_m) v_m = 0,
  \end{equation}
  which by linear independence forces $a_1 = c_1, \dots, a_m = c_m$.

  Note: The empty list $()$ is also defined to be linearly independent.
\end{mydef}

\begin{mydef} [linear dependence]
  A list of vectors in $V$ is called \qt{linearly dependent} if it is not linearly independent.
  That is, $\oneTillM{v}$ is linearly dependent if there exist scalars $a_1, \dots, a_m \in \myF$, not all zero, such that
  \begin{equation}
    a_1 v_1 + \dots + a_m v_m = 0.
  \end{equation}
  For example, if two non-zero vectors $u, v \in V$ are linearly dependent, there exist $a, b \in \myF$ with $(a,b) \neq (0,0)$ such that $au + bv = 0$. Since $u, v \neq 0$, both $a \neq 0$ and $b \neq 0$, so $u = \left( -\sfrac{b}{a} \right) v$.
\end{mydef}

\setcounter{thm}{18}
\begin{thm} [linear dependence lemma]
  \label{thm: linear dependence lemma}
  Suppose the list $v_{1}, \dots, v_{m}\in V$ is linearly dependent. Then there is an index $k \in \{ 1, \dots, m \}$ with
  \begin{equation}
    \label{eq: conditions in the linear dependence lemma}
    v_k \in \myspan {v_1, \dots, v_{k-1}}.
  \end{equation}
  Furthermore, if $k$ satisfies the condition above \eqref{eq: conditions in the linear dependence lemma} and $v_k$ is removed from $v_1, \dots, v_m$, then the span of the remaining list equals $\myspan {v_1, \dots, v_{m}}$:
  \begin{equation}
    \myspan {v_1, \dots, v_{k-1}, \xcancel{v_k}, v_{k+1}, \dots, v_m} = \myspan {v_1, \dots, v_m}.
  \end{equation}
\end{thm}

\begin{prf}
  Because the list $\onetillm{v}$ is linearly dependent, there exist scalars $a_1, \ldots, a_m \in \myF$, not all zero, such that
  \begin{equation}
    a_1 v_1 + \cdots + a_m v_m = 0.
  \end{equation}
  Let $k$ be the largest element of $\setonetillm$ such that $a_k \neq 0$.
  If $k = 1$, then $a_1 v_1 = 0$, so $v_1 = 0 \in \myspan{\;} = \{0\}$.
  If $k > 1$, then dividing by $a_k \neq 0$ gives
  \begin{equation}
    v_k = - \frac{a_1}{a_k}v_1 - \cdots - \frac{a_{k-1}}{a_k}v_{k-1},
  \end{equation}
  showing that $v_k \in \myspan{v_1, \ddd, v_{k-1}}$.

  Now for the second claim, suppose $k \in \setonetillm$ satisfies $v_k \in \myspan{v_1, \ddd, v_{k-1}}$, so
  \begin{equation}
    \label{eq: equation for v_k}
    v_k = b_1v_1 + \cdots + b_{k-1}v_{k-1} \quad \text{for some } b_1, \ddd, b_{k-1} \in \myF.
  \end{equation}
  Let $u \in \myspan{v_1, \ddd, v_m}$. Then there exist $c_1, \ddd, c_m \in \myF$ such that
  \begin{equation}
    \label{eq: first linear combination for u}
    u = c_1v_1 + \cdots + c_kv_k + \cdots + c_{m}v_{m}.
  \end{equation}
  Substituting \eqref{eq: equation for v_k} into \eqref{eq: first linear combination for u} gives
  \begin{equation}
    \begin{aligned}
      \label{eq: second linear combination for u}
      u &= c_1v_1 + \cdots + c_k (b_1v_1 + \cdots + b_{k-1}v_{k-1}) + c_{k+1} v_{k+1} + \cdots + c_m v_m \\
        &= (c_1 + c_k b_1) v_1 + \cdots + (c_{k-1}+c_k b_{k-1}) v_{k-1} + c_{k+1} v_{k+1} + \cdots + c_m v_m,
    \end{aligned}
  \end{equation}
  which shows that $u \in \myspan{v_1, \ddd, v_{k-1}, v_{k+1}, \ddd, v_m}$. Hence,
  \begin{equation}
    \myspan{v_1, \ddd, v_{k-1}, \xcancel{v_k}, v_{k+1}, \ddd, v_m} = \myspan{v_1, \ddd, v_m}.
  \end{equation}
\end{prf}

\setcounter{thm}{21}
\begin{thm} [length of linearly independent list $\le$ length of spanning list]
  \label{thm: length of linearly independent list at most length of spanning list}
  In a \fd vector space, the length of every linearly independent list of vectors is less than or equal to the length of every spanning list of vectors.
\end{thm}
\begin{prf}
  Let $u_1, \ldots, u_m \in V$ be a linearly independent list and let $w_1, \ldots, w_n \in V$ be a spanning list of $V$, so $V = \myspan{w_1, \ldots, w_n}$.
  We need to prove that $m \leq n$.
  
  We construct, in $m$ steps, a spanning list of size $n$ that incorporates $u_1,\dots,u_m$ one by one, replacing one of the $w$'s at each step.
  Start with
  \[
      B_0 := (w_1, \dots, w_n), \quad V = \myspan{B_0}.
  \]

  \prooffont{Step 1:} Add $u_1$ to $B_0$, so set
  \begin{equation}
    B_1^* := (u_1, w_1, \ldots, w_n).
  \end{equation}

  Since $u_1 \in \myspan {B_0}$, the enlarged list is linearly dependent. By the Linear Dependence Lemma \ref{thm: linear dependence lemma}, one of the vectors in $B_1^*$ is in the span of the previous vectors. Since $u_1 \neq 0$ (as $u_1, \ddd, u_m$ is linearly independent), that vector cannot be $u_1$; it must be one of the original $w$'s, say $w_j$. Removing $w_j$ gives
  \begin{equation*}
    B_1 := B_1^* \setminus \{w_j\} = \Big(u_1, w_1, \ldots, w_{j-1}, \xcancel{w_j}, w_{j+1}, \ldots, w_n\Big), \quad V=\myspan {B_1}.
  \end{equation*}

  \prooffont{Step k, for k=2$\mathbf{, \ddd,}$m:} The list $B_{k-1}$ from the previous step $k-1$ spans $V$. So $u_{k} \in \myspan{B_{k-1}}$. Adding $u_k$ to the list makes it linearly dependent. Let
  \begin{equation*}
    B_k^* :\equiv \Big(u_1, \ldots, u_{k-1}, u_k, \underbrace{\widetilde w_{1}, \ldots, \widetilde w_{n-(k-1)} }_{\text{remaining $w$'s}} \Big), \quad V=\myspan {B_k^*}.
  \end{equation*}

  By the Linear Dependence Lemma, some vector in $B_k^*$ is in the span of the preceding vectors. It cannot be any of the $u$'s (since $u_1, \ddd, u_m$ is linearly independent), so it must be one of the remaining $w$'s, say $\widetilde w_j$, obtaining
  \begin{equation*}
    B_k := B_k^* \setminus \{\widetilde w_j \} = \Big(u_1, \ldots, u_{k}, \widetilde w_1, \ldots, \widetilde w_{j-1}, \xcancel{\widetilde w_j}, \widetilde w_{j+1}, \ldots, \widetilde w_{n-(k-1)}\Big), \quad V=\myspan B_k.
  \end{equation*}

  After each step $k$, $|B_k| = n$ and $u_1,\dots,u_k\in B_k$.

  \emph{\bfseries Conclusion:} After $m$ steps, $B_m$ is a spanning list of size $n$ containing the $m$ independent vectors; hence, $m \leq n$.
\end{prf}

\setcounter{thm}{24}
\begin{thm} [finite-dimensional subspace]
  \label{thm: finite-dimensional subspace}
  Every subspace of a finite\-/dimensional vector space is finite\-/dimensional.
\end{thm}
\begin{prf}
  Suppose $\dim V<\infty$, and $U\subseteq V$. Let $m := \dim V$.

  \emph{Step 1:} If
  \begin{equation}
    U = \{0\},
  \end{equation}
  we are done. Otherwise, pick any nonzero $u_1 \in U$.

  \prooffont{Step k, ($\mathbf{2 \le}$ k $\mathbf{\le m}$):} If
  \begin{equation}
    U = \myspan{u_1, \ldots, u_{k-1}},
  \end{equation}
  then $U$ is finite-dimensional and we are done.

  If
  \begin{equation}
    U \neq \myspan{u_1, \ldots, u_{k-1}},
  \end{equation}
  then choose a vector $u_k \in U$ \st
  \begin{equation}
    u_k \notin \myspan{u_1, \ldots, u_{k-1}}.
  \end{equation}

  After each step, as long as the process continues, we have constructed a list of vectors such that no vector in the list is in the span of the previous vectors. Hence, at each stage we maintain a linearly independent list, by the Linear Dependence Lemma (\ref{thm: linear dependence lemma}). No list of independent vectors in $V$ can exceed the length of any spanning list of $V$, by \ref{thm: length of linearly independent list at most length of spanning list}.
  Thus, the process eventually terminates (after at most $m=\dim V$ steps), so $U$ has a finite spanning list and is therefore finite-dimensional.
\end{prf}

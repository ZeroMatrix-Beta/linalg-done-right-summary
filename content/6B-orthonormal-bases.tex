\section{Orthonormal Bases}

\subsection{Orthonormal Lists and the Gram-Schmidt Procedure}
\mce{22}
\begin{mydef}[orthonormal]
  A list of vectors is called \qt{orthonormal} if each vector in the list has norm $1$ and is orthogonal to all the other vectors in the list. Since the square root of $1$ is $1$, we can use the following definition as well:

  A list $e_1, \ddd, e_m$ of vectors in $V$ is \qt{orthonormal} if, for all $j,k \in \{1, \ddd, m\}$,
  \begin{equation}
    \ip{e_j}{e_k} = \begin{cases}
      1,  & \text{if $j=k$} \\
      0, & \text{if $j \neq k$}.
    \end{cases}
  \end{equation}

  Recall that $\norm{v} \equiv \sqrt{\ip{v}{v}}$.
\end{mydef}

\mce{24}
\begin{thm}[norm of an orthonormal linear combination]
  \label{thm: norm of an orthonormal linear combination}
  If $e_1, \ddd, e_m$ is an orthonormal list of vectors in $V$, then
  \begin{equation}
    \norm{a_1e_1 + \cdots+ a_m e_m} ^2 = |a_1|^2 + \cdots + |a_m|^2 \quad \forall a_1, \ddd, a_m \in \myF.
  \end{equation}
\end{thm}
\begin{prf}
  Because $\forall k: \norm{e_k}=1$, this follows from repeated applications of the Pythagorean theorem \ref{thm: pythagorean theorem}:
  \begin{equation}
    \begin{aligned}
      \norm{a_1e_1 + \cdots+ a_m e_m} ^2
      & = \norm{(a_1e_1 + \cdots + a_{m-1}e_{m-1}) + a_m e_m} ^2 \\
      & = \norm{(a_1e_1 + \cdots + a_{m-1}e_{m-1})}^2 + \norm{a_m e_m} ^2 \\
      & = \norm{(a_1e_1 + \cdots + a_{m-2}e_{m-2}) + a_{m-1}e_{m-1}}^2 + |a_m|^2 \\
      & = \norm{(a_1e_1 + \cdots + a_{m-2}e_{m-2})}^2 + \norm{a_{m-1}e_{m-1}}^2 + |a_m|^2\\
      & = \norm{(a_1e_1 + \cdots + a_{m-3}e_{m-3}) + a_{m-2}e_{m-2}}^2 + |a_{m-1}|^2 + |a_m|^2\\
      & = \; \cdots \\
      & = |a_1|^2 + |a_2|^2 + \cdots + |a_{m-1}|^2 + |a_m|^2.
    \end{aligned}
  \end{equation}
  \vspace{-1em}
\end{prf}

\mce{25}
\begin{thm}[linear independence]
  \label{thm: orthonormal lists are linearly independent}
  Every orthonormal list of vectors is linearly independent.
\end{thm}
\begin{prf}
  Suppose $e_1, \ddd, e_m$ is an orthonormal list of vectors in $V$ and $a_1, \ddd, a_m \in \myF$ are \st
  \begin{equation}
    a_1e_1 + \cdots + a_m e_m = 0.
  \end{equation}

  This implies that $|a_1|^2 + \cdots + |a_m|^2 = 0$ by \ref{thm: norm of an orthonormal linear combination}, which means that all the $a_k$'s are $0$, which is what we needed to show.
\end{prf}

\mce{26}
\begin{thm}[Bessel's inequality]
  \label{thm: Bessel's inequality}
  Suppose $e_1, \ddd, e_m$ is an orthonormal list of vectors in $V$. If $v \in V$, then
  \begin{equation}
    |\ip{v}{e_1}|^2 + \cdots + |\ip{v}{e_m}|^2 \leq \norm{v}^2.
  \end{equation}
\end{thm}
\begin{prf}
  Suppose $v \in V$. Then
  \begin{equation}
    v= \underbrace{\ip{v}{e_1}e_1 + \cdots + \ip{v}{e_m} e_m}_u + \underbrace{\left(v - (\ip{v}{e_1}e_1 + \cdots + \ip{v}{e_m} e_m)\right)}_w.
  \end{equation}

  Let $u$ and $w$ be defined as in the equation above. If $k \in \{1, \ddd, m\}$, then
  \begin{equation}
    \begin{aligned}
      \ip{w}{e_k}
      &=\ip{v - \left(\underbrace{\ip{v}{e_1}}_{\in \; \myF}e_1 + \cdots + \underbrace{\ip{v}{e_m}}_{\in \; \myF} e_m\right)}{ \,e_k \,} \\
      &=\ip{v}{e_k} - \ip{v}{e_1}\ip{e_1}{e_k} - \ip{v}{e_2}\ip{e_2}{e_k} - \, \cdots \, - \ip{v}{e_m}\ip{e_m}{e_k} \\
      &=\ip{v}{e_k} - \ip{v}{e_k}\ip{e_k}{e_k} \\
      &=0. \mytext{Hence,} \\
      \ip{w}{u}
      & = \ip{w}{\underbrace{\ip{v}{e_1}}_{\in \; \myF}e_1 + \cdots + \underbrace{\ip{v}{e_m}}_{\in \; \myF} e_m} \\
      & = \ip{w}{\ip{v}{e_1}e_1} + \cdots + \ip{w}{\ip{v}{e_m}e_m} \\
      & = \overline{\ip{v}{e_1}}\ip{w}{e_1} + \cdots + \overline{\ip{v}{e_m}}\ip{w}{e_m} \\
      & = 0.
    \end{aligned}
  \end{equation}

  Since this means that $u$ and $w$ are orthogonal, we can use the well-known Pythagorean theorem \ref{thm: pythagorean theorem} to conclude that
  \begin{equation}
    \begin{aligned}
      \norm{v}^2
      &=    \norm{u}^2 + \norm{w}^2 \\
      &\geq \norm{u}^2 \\
      &= |\ip{v}{e_1}|^2 + \cdots + |\ip{v}{e_m}|^2,
    \end{aligned}
  \end{equation}
  where the last line comes from \ref{thm: norm of an orthonormal linear combination}. If you look at the definition of $u=\ip{v}{e_1}e_1 + \cdots + \ip{v}{e_m} e_m$, the inner products are the coefficients with respect to the orthonormal list $e_1, \ddd, e_m$.
\end{prf}

\begin{mydef}[orthonormal basis]
  An \qt{orthonormal} basis of $V$ is an orthonormal list of vectors in $V$ that is also a basis of $V$.
\end{mydef}

\begin{thm}[orthonormal lists of the right length are orthonormal bases]
  \label{thm: orthonormal lists of the right length are orthonormal bases}
  Suppose $V$ is finite-dimensional. Then every orthonormal list of vectors in $V$ of length $\dim V$ is an orthonormal basis of $V$.
\end{thm}
\begin{prf}
  Every orthonormal list is linearly independent by \ref{thm: orthonormal lists are linearly independent}. By \ref{thm: linearly independent list of the right length is a basis}, every linearly independent list in $V$ of length $\dim V$ is a basis of $V$. Hence, any such orthonormal list is an orthonormal basis of $V$.
\end{prf}

\mce{30}
\begin{thm}[writing a vector as a linear combination of an orthonormal basis]
  \label{thm: writing a vector as a linear combination of an orthonormal basis}

  Suppose $e_1, \ddd, e_n$ is an orthonormal basis of $V$ and $u,v \in V$. Then:
  \begin{enumerate}[label=\textbf{(\alph*)}]
    \item $v=\ip{v}{e_1}e_1 + \cdots + \ip{v}{e_n}e_n$;
    \item $\norm{v}^2 = |\ip{v}{e_1}|^2 + \cdots + |\ip{v}{e_n}|^2$;
    \item $\ip{u}{v} = \ip{u}{e_1} \overline{\ip{v}{e_1}} + \cdots + \ip{u}{e_n} \overline{\ip{v}{e_n}}$.
  \end{enumerate}
\end{thm}
\begin{prf}
  Let $v=a_1 e_1 + \cdots + a_n e_n$. Then, $\forall k \in \{1, \ddd, n\}$, we have
  \begin{equation}
    \begin{aligned}
      \ip{v}{e_k}
      &=\ip{a_1 e_1 + \cdots + a_n e_n}{e_k} \\
      &=\ip{a_1 e_1}{e_k} + \cdots + \ip{a_n e_n}{e_k}\\
      &=\ip{a_ke_k}{e_k} \\
      &= a_k\ip{e_k}{e_k} \\
      &=a_k.
    \end{aligned}
  \end{equation}
  Thus, \textbf{(a)} holds.

  Now, \textbf{(b)} follows immediately from \textbf{(a)} and \ref{thm: norm of an orthonormal linear combination}.

  Using conjugate linearity,
  \begin{equation}
    \begin{aligned}
      \ip{u}{v}
      &=\ip{u}{\ip{v}{e_1}e_1 + \cdots + \ip{v}{e_n}e_n} \\
      &= \ip{u}{\ip{v}{e_1}e_1} + \cdots + \ip{u}{\ip{v}{e_n}e_n} \\
      &= \overline{\ip{v}{e_1}}\ip{u}{e_1} + \cdots + \overline{\ip{v}{e_n}}\ip{u}{e_n} \\
      &= \ip{u}{e_1}\overline{\ip{v}{e_1}} + \cdots + \ip{u}{e_n}\overline{\ip{v}{e_n}}.
    \end{aligned}
  \end{equation}

  Therefore, \textbf{(c)} holds.
\end{prf}

\mce{31}
\begin{thm}[Gram-Schmidt procedure]
  \label{thm: gram-schmidt procedure}
  Suppose $v_1, \ddd, v_m$ is a linearly independent list of vectors in $V$. Let $e_1 := \frac{v_1}{\norm{v_1}}$. For $k = 2, \ddd, m$, define $e_k$ inductively by
  \begin{equation}
    e_k := \frac{v_k - \sum_{j=1}^{k-1} \ip{v_k}{e_j} e_j}{\norm{v_k - \sum_{j=1}^{k-1} \ip{v_k}{e_j} e_j}}.
  \end{equation}
  Then $e_1, \ddd, e_m$ is an orthonormal list of vectors in $V$ such that
  \begin{equation}
    \myspan{v_1, \ddd, v_k} = \myspan{e_1, \ddd, e_k} \quad \text{for each } k \in \{1, \ddd, m\}.
  \end{equation}
\end{thm}
\begin{prf}
  We proceed by induction on $k$. For $k=1$, $\norm{e_1} = \norm{\frac{v_1}{\norm{v_1}}} = 1$, and $\myspan{e_1} = \myspan{v_1}$.
  Suppose $1 < k \leq m$ and $e_1, \ddd, e_{k-1}$ is an orthonormal list whose span equals $\myspan{v_1, \ddd, v_{k-1}}$.
  Since $v_1, \ddd, v_m$ is linearly independent, $v_k \notin \myspan{v_1, \ddd, v_{k-1}} = \myspan{e_1, \ddd, e_{k-1}}$.
  Thus, the numerator $v_k - \sum_{j=1}^{k-1} \ip{v_k}{e_j} e_j$ is nonzero, so $e_k$ is well-defined and has norm $1$.
  For each $1 \leq \ell < k$:
  \begin{equation}
    \ip{e_k}{e_\ell} = \frac{\ip{v_k}{e_\ell} - \sum_{j=1}^{k-1} \ip{v_k}{e_j} \ip{e_j}{e_\ell}}{\norm{v_k - \sum_{j=1}^{k-1} \ip{v_k}{e_j} e_j}} = \frac{\ip{v_k}{e_\ell} - \ip{v_k}{e_\ell}}{\norm{v_k - \sum_{j=1}^{k-1} \ip{v_k}{e_j} e_j}} = 0.
  \end{equation}
  Thus, $e_1, \ddd, e_k$ is an orthonormal list.
  Since $e_k \in \myspan{v_1, \ddd, v_k}$ and $v_k \in \myspan{e_1, \ddd, e_k}$, we have $\myspan{v_1, \ddd, v_k} = \myspan{e_1, \ddd, e_k}$.
\end{prf}

\mce{34}
\begin{thm}[existence of orthonormal basis]
  \label{thm: existence of orthonormal basis}
  Every finite-dimensional inner product space has an orthonormal basis.
\end{thm}
\begin{prf}
  Let $V$ be a finite-dimensional inner product space. By \ref{thm: every finite-dimensional vector space has a basis}, $V$ has a basis $v_1, \ddd, v_n$. Applying the Gram-Schmidt Procedure (\ref{thm: gram-schmidt procedure}) to $v_1, \ddd, v_n$ produces an orthonormal list $e_1, \ddd, e_n$ in $V$ of length $n = \dim V$. By \ref{thm: orthonormal lists are linearly independent} it is linearly independent, so by \ref{thm: linearly independent list of the right length is a basis}, $e_1, \ddd, e_n$ is an orthonormal basis of $V$.
\end{prf}

\mce{38}
\begin{thm}[Schur's theorem]
  \label{thm: schur's theorem}
  Suppose $V$ is a finite-dimensional complex inner product space and $T \in \linmap(V)$. Then there is an orthonormal basis of $V$ with respect to which $T$ has an upper-triangular matrix.
\end{thm}
\begin{prf}
  By \ref{thm: necessary and sufficient condition to have an upper-triangular-matrix}, there is a basis $v_1, \ddd, v_n$ of $V$ with respect to which $T$ has an upper-triangular matrix.
  Apply the Gram-Schmidt Procedure to $v_1, \ddd, v_n$ to obtain an orthonormal basis $e_1, \ddd, e_n$.
  Since $\myspan{e_1, \ddd, e_k} = \myspan{v_1, \ddd, v_k}$ for each $k \in \{1, \ddd, n\}$, and each subspace $\myspan{v_1, \ddd, v_k}$ is invariant under $T$, each subspace $\myspan{e_1, \ddd, e_k}$ is invariant under $T$.
  Thus, by \ref{thm: conditions for upper-triangular matrix}, the matrix of $T$ with respect to $e_1, \ddd, e_n$ is upper triangular.
\end{prf}

\subsection{Linear Functionals on Inner Product Spaces}

\mce{42}
\begin{thm}[Riesz representation theorem]
  \label{thm: riesz representation theorem}
  Suppose $V$ is a finite-dimensional inner product space and $\varphi \in \dual{V}$ is a linear functional on $V$. Then there exists a unique vector $u \in V$ such that
  \begin{equation}
    \varphi(v) = \ip{v}{u} \quad \forall v \in V.
  \end{equation}
\end{thm}
\begin{prf}
  \prooffont{Existence:} Let $e_1, \ddd, e_n$ be an orthonormal basis of $V$.
  For any $v \in V$, by \ref{thm: writing a vector as a linear combination of an orthonormal basis}(a), $v = \sum_{k=1}^n \ip{v}{e_k} e_k$.
  Applying $\varphi$ gives
  \begin{equation}
    \varphi(v) = \sum_{k=1}^n \ip{v}{e_k} \varphi(e_k) = \left\langle v, \, \sum_{k=1}^n \overline{\varphi(e_k)} e_k \right\rangle.
  \end{equation}
  Setting $u := \sum_{k=1}^n \overline{\varphi(e_k)} e_k \in V$, we have $\varphi(v) = \ip{v}{u}$ for all $v \in V$.

  \prooffont{Uniqueness:} Suppose $u, \widetilde{u} \in V$ satisfy $\ip{v}{u} = \varphi(v) = \ip{v}{\widetilde{u}}$ for all $v \in V$.
  Then $\ip{v}{u - \widetilde{u}} = 0$ for all $v \in V$. Taking $v = u - \widetilde{u}$ gives $\norm{u - \widetilde{u}}^2 = 0 \implies u = \widetilde{u}$.
\end{prf}

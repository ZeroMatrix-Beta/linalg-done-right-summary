\section*{Exercises about Products and Quotients of Vector Spaces}

\exercise{3}
\begin{xrcs}
  Suppose $V_1, \ddd, V_m$ are vector spaces. Prove that $\linmap(V_1 \times \dots \times V_m, W)$ and $\linmap(V_1, W) \times \dots \times \linmap(V_m, W)$ are isomorphic vector spaces.

\begin{xprf}
  Define a map $\Psi: \linmap(V_1 \times \dots \times V_m, W) \to \linmap(V_1, W) \times \dots \times \linmap(V_m, W)$ by
  \[
    \Psi(T) = (T_1, \dots, T_m), \quad \text{where } T_k(v) = T(0, \dots, 0, v, 0, \dots, 0) \text{ for } v \in V_k.
  \]
  Each $T_k \in \linmap(V_k, W)$ is linear, and $\Psi$ is clearly linear.
  To show $\Psi$ is an isomorphism, define the inverse map $\Phi: \linmap(V_1, W) \times \dots \times \linmap(V_m, W) \to \linmap(V_1 \times \dots \times V_m, W)$ by
  \[
    \Phi(S_1, \dots, S_m)(v_1, \dots, v_m) = S_1(v_1) + \dots + S_m(v_m).
  \]
  Then $\Phi(S_1, \dots, S_m)$ is a linear map, and it is straightforward to verify that $\Psi \circ \Phi = I$ and $\Phi \circ \Psi = I$.
  Thus $\Psi$ is an isomorphism.
\end{xprf}
\end{xrcs}

\exercise{6}
\begin{xrcs}
  Suppose that $v, x \in V$ and that $U, W$ are subspaces of $V$ such that $v + U = x + W$. Prove that $U = W$.
  \begin{xprf}
    Since $v \in v + U$ and $v+U = x + W$, it follows that $v \in x + W$. Hence, there is  $w_0 \in W$ with $v = x + w_0$, so $(x-v) = -w_0 \in W$. Now,
    \begin{equation}
      U = (-v) + (v+U) = (-v) + (x + W) = (x-v) + W = W,
    \end{equation}

    where the last equality uses $x-v \in W$ and the fact that translating a subspace by one of its own elements leaves it unchanged (i.e., $(x-v) + W = W$). Hence, $U = W$.
  \end{xprf}
\end{xrcs}

\exercise{10}
\begin{xrcs}
  Suppose $A_1 = v + U_1$ and $A_2 = w + U_2$ for some $v, w \in V$ and some subspaces $U_1, U_2$ of $V$. Prove that the intersection $A_1 \cap A_2$ is either a translate of some subspace of $V$ or is the empty set.
  \begin{xprf}
    By the definition of intersection, we get
    \begin{equation}
      A_1 \cap A_2 = \{ x \in V : x = v + u_1 \text{ and } x = w + u_2, \; u_1 \in U_1, \; u_2 \in U_2 \}.
    \end{equation}

    The condition $v + u_1 = w + u_2$ is equivalent to
    \begin{equation}
      v-w = u_2 - u_1, u_1 \in U_1, u_2 \in U_2 \quad \Longleftrightarrow \quad v-w \in U_1 + U_2.
    \end{equation}

    If $v-w \notin U_1 + U_2$, there is no solution for $u_1$ and $u_2$ and $A_1 \cap A_2 = \varnothing$. If a solution exists, put
    \begin{equation}
      \begin{aligned}
        r^0 &:= v + u_1^0  \\
            &:= w + u_2^0,
      \end{aligned}
    \end{equation}

    where $r^0 \in A_1 \cap A_2$ and for appropriate $u_1^0 \in U_1, u_2^0 \in U_2$ (i.e., $v-w = u_2^0 - u_1^0$).
    Define
    \begin{equation}
      S := \{ r^0 + u : u \in U_1 \cap U_2 \}.
    \end{equation}

    We want to show that $A_1 \cap A_2 = S$. Therefore, take any $x \in A_1 \cap A_2$, so that $x = v + u_1$ or $x = w + u_2$ with $u_1 \in U_1, u_2 \in U_2$. Now, it follows that
    \begin{equation}
      \begin{aligned}
        x-r^0 &= (v + u_1) \,- (v + u_1^0) \, = u_1 - u_1^0 \in U_1,  \text{ and } \\
        x-r^0 &= (w + u_2)   - (w + u_2^0)    = u_2 - u_2^0 \in U_2, \\
      \end{aligned}
    \end{equation}

    and therefore, $x-r^0 \in U_1 \cap U_2$. Since $x = r^0 + (x-r^0)$, it is immediate that
    \begin{equation}
      x \in r^0 + (U_1 \cap U_2).
    \end{equation}

    Hence, $A_1 \cap A_2 \subseteq S$. Conversely, pick any $u \in U_1 \cap U_2$, so that $r^0 + u \in S$. Then
    \begin{equation}
      \begin{aligned}
        r^0 + u &= (v + u_1^0) + u = v + ( u_1^0 + u) \in v + U_1 = A_1, \text{ and } \\
        r^0 + u &= (w + u_2^0) + u = w + ( u_2^0 + u) \in w + U_2 = A_2.  \\
      \end{aligned}
    \end{equation}

    Thus, $S \subseteq A_1 \cap A_2$. Therefore, we have shown that
    \begin{equation}
      A_1 \cap A_2 =
      \begin{cases}
        S,            & \text{ if } v-w \in U_1 + U_2, \\
        \varnothing,  & \text{ otherwise. }
      \end{cases}
    \end{equation}
  \end{xprf}
\end{xrcs}

\exercise{13}
\begin{xrcs}
  Suppose $T \in \linmap(V,W)$. Define $\widetilde{T}: V/\mynull T \to W$ by
  \begin{equation}
    \widetilde{T}(v + \mynull T) := Tv.
  \end{equation}
  Prove that $\widetilde{T}$ is an isomorphism from $V/\mynull T$ onto $\myrange T$ (the First Isomorphism Theorem for Vector Spaces).
\begin{xprf}
  \StepOne We first show that $\widetilde{T}$ is well-defined and injective.
  Let $u, v \in V$. Then:
  \[
    u + \mynull T = v + \mynull T \iff u - v \in \mynull T \iff T(u-v) = 0 \iff Tu = Tv.
  \]
  The forward direction ($\implies$) shows that $\widetilde{T}$ is well-defined.
  The backward direction ($\impliedby$) shows that $\widetilde{T}(u + \mynull T) = \widetilde{T}(v + \mynull T) \implies u + \mynull T = v + \mynull T$, so $\widetilde{T}$ is injective.

  \StepTwo We verify linearity. For any $u, v \in V$ and $\lambda \in \myF$:
  \[
  \begin{aligned}
    \widetilde{T}((u + \mynull T) + (v + \mynull T)) &= \widetilde{T}((u+v) + \mynull T) = T(u+v) = Tu + Tv \\
    &= \widetilde{T}(u + \mynull T) + \widetilde{T}(v + \mynull T),
  \end{aligned}
  \]
  and $\widetilde{T}(\lambda(u + \mynull T)) = \widetilde{T}(\lambda u + \mynull T) = T(\lambda u) = \lambda Tu = \lambda \widetilde{T}(u + \mynull T)$.

  \StepThree Finally, we check that $\widetilde{T}$ is surjective onto $\myrange T$.
  Let $w \in \myrange T$. By definition of the range, there exists $v \in V$ such that $Tv = w$.
  Then $\widetilde{T}(v + \mynull T) = Tv = w$.
  Thus, $\widetilde{T}$ is a linear bijection from $V/\mynull T$ onto $\myrange T$, hence an isomorphism.
\end{xprf}
\end{xrcs}


\exercise{14}
\begin{xrcs}
  Suppose $U$ and $W$ are subspaces of $V$ and $V = U \oplus W$. Suppose $w_1, \ddd, w_m$ is a basis of $W$. Prove that
  \begin{equation}
    w_1 + U, \ddd, w_m + U \text{ is a basis of } V/U.
  \end{equation}

\begin{xprf}
  \StepOne We show spanning. Let $v + U \in V/U$. Since $V = U \oplus W$, $v = u + w$ for some $u \in U, w \in W$.
  Since $w_1, \dots, w_m$ is a basis of $W$, $w = c_1 w_1 + \dots + c_m w_m$ for some scalars $c_j$.
  Then $v + U = (u + w) + U = w + U = (c_1 w_1 + \dots + c_m w_m) + U = c_1(w_1+U) + \dots + c_m(w_m+U)$.
  Thus $w_1+U, \dots, w_m+U$ spans $V/U$.

  \StepTwo We show linear independence. Suppose $c_1(w_1+U) + \dots + c_m(w_m+U) = 0 + U$.
  Then $(c_1 w_1 + \dots + c_m w_m) + U = 0 + U$, which implies $c_1 w_1 + \dots + c_m w_m \in U$.
  Since each $w_j \in W$, $c_1 w_1 + \dots + c_m w_m \in W$.
  Thus $c_1 w_1 + \dots + c_m w_m \in U \cap W = \{0\}$, so $c_1 w_1 + \dots + c_m w_m = 0$.
  Since $w_1, \dots, w_m$ is a basis of $W$, $c_1 = \dots = c_m = 0$.
  Therefore, $w_1+U, \dots, w_m+U$ is linearly independent, hence a basis of $V/U$.
\end{xprf}
\end{xrcs}

\exercise{15}
\begin{xrcs}
  Suppose $U$ is a subspace of $V$, $v_1 + U, \ddd, v_m + U$ is a basis of $V/U$, and $u_1, \ddd, u_n$ is a basis of $U$. Prove that
  \begin{equation}
    v_1, \ddd, v_m, u_1, \ddd, u_n \text{ is a basis of } V.
  \end{equation}

\begin{xprf}
  \StepOne We show spanning. Let $v \in V$.
  Since $v_1+U, \dots, v_m+U$ is a basis of $V/U$, there exist scalars $a_1, \dots, a_m$ such that $v + U = a_1(v_1+U) + \dots + a_m(v_m+U) = (a_1 v_1 + \dots + a_m v_m) + U$.
  This means $v - (a_1 v_1 + \dots + a_m v_m) \in U$.
  Since $u_1, \dots, u_n$ is a basis of $U$, there exist scalars $b_1, \dots, b_n$ such that $v - \sum_{j=1}^m a_j v_j = \sum_{k=1}^n b_k u_k$.
  Thus $v = a_1 v_1 + \dots + a_m v_m + b_1 u_1 + \dots + b_n u_n$.
  Therefore, $v_1, \dots, v_m, u_1, \dots, u_n$ spans $V$.

  \StepTwo We show linear independence. Suppose $a_1 v_1 + \dots + a_m v_m + b_1 u_1 + \dots + b_n u_n = 0$.
  Applying the quotient map $\pi: V \to V/U$ gives $a_1(v_1+U) + \dots + a_m(v_m+U) + 0 = 0 + U$.
  Since $v_1+U, \dots, v_m+U$ is a basis of $V/U$, $a_1 = \dots = a_m = 0$.
  The equation then reduces to $b_1 u_1 + \dots + b_n u_n = 0$.
  Since $u_1, \dots, u_n$ is a basis of $U$, $b_1 = \dots = b_n = 0$.
  Thus all coefficients are zero, which proves linear independence.
  Therefore, the list is a basis of $V$.
\end{xprf}
\end{xrcs}

\section*{Exercises about Matrices}

\exercise{1}
\begin{xrcs}
  Suppose $V$ and $W$ are finite-dimensional and $T \in \linmap(V,W)$. Prove that with respect to any bases of $V$ and $W$,
  \begin{equation}
    \dim \myrange T = \text{column rank of } \mmatrix(T).
  \end{equation}

  \begin{xprf}
    Let $v_1, \dots, v_n$ be a basis of $V$ and $w_1, \dots, w_m$ be a basis of $W$.
    Since $v_1, \dots, v_n$ spans $V$, we have $\myrange T = \myspan{Tv_1, \dots, Tv_n}$.
    Define the coordinate isomorphism $\Phi : W \to \myF^{m,1}$ with respect to the basis $w_1, \dots, w_m$ by
    \begin{equation}
      \Phi\left(\sum_{j=1}^m c_j w_j\right) := (c_1, \dots, c_m)^\top.
    \end{equation}
    For each $k \in \{1, \dots, n\}$, by definition of the matrix $\mmatrix(T)$, $\Phi(Tv_k) = \mmatrix(T)_{\mathsmaller{\bullet}, k}$ (the $k$-th column of $\mmatrix(T)$).
    Since $\Phi$ is an isomorphism, it preserves dimensions of subspaces:
    \begin{equation}
      \dim \myrange T = \dim \myspan{Tv_1, \dots, Tv_n} = \dim \myspan{\mmatrix(T)_{\mathsmaller{\bullet}, 1}, \dots, \mmatrix(T)_{\mathsmaller{\bullet}, n}} = \text{column rank of } \mmatrix(T).
    \end{equation}
  \end{xprf}
\end{xrcs}

\exercise{2}
\begin{xrcs}
  Suppose $A \in \myF^{m,n}$ and $C \in \myF^{n,p}$. Prove that
  \begin{equation}
    \operatorname{rank}(AC) \leq \min\{\operatorname{rank} A, \operatorname{rank} C\}.
  \end{equation}

  \begin{xprf}
    By column-wise matrix multiplication, each column of $AC$ is a linear combination of the columns of $A$. Therefore, the span of the columns of $AC$ is contained in the span of the columns of $A$, which implies
    \begin{equation}
      \operatorname{rank}(AC) \leq \operatorname{rank} A.
    \end{equation}
    Similarly, by row-wise matrix multiplication, each row of $AC$ is a linear combination of the rows of $C$. Therefore, the span of the rows of $AC$ is contained in the span of the rows of $C$, which implies
    \begin{equation}
      \text{row rank of } (AC) \leq \text{row rank of } C.
    \end{equation}
    Since the row rank of any matrix equals its column rank (Theorem \ref{thm: column rank equals row rank}), this gives $\operatorname{rank}(AC) \leq \operatorname{rank} C$.
    Combining both inequalities yields $\operatorname{rank}(AC) \leq \min\{\operatorname{rank} A, \operatorname{rank} C\}$.
  \end{xprf}
\end{xrcs}
\exercise{6}
\begin{xrcs}
  Suppose $A \in \myF^{m,n}$. Prove that $\operatorname{rank} A = 1$ if and only if there exist nonzero $c \in \myF^{m,1}$ and $d \in \myF^{1,n}$ such that
  \begin{equation}
    A = c d.
  \end{equation}

\begin{xprf}
  \Rightarrowdirection Suppose $\operatorname{rank} A = 1$. The column space of $A$ has dimension $1$.
  Choose a nonzero column $c \in \myF^{m,1}$ of $A$. Then $c$ is a basis of the column space of $A$.
  Thus every column $A_{\mathsmaller{\bullet}, k}$ of $A$ is a scalar multiple of $c$, so there exists $d_k \in \myF$ such that $A_{\mathsmaller{\bullet}, k} = d_k c = c d_k$ for each $k \in \{1, \dots, n\}$.
  Defining the row vector $d = (d_1, \dots, d_n) \in \myF^{1,n}$, we have $A = c d$.
  Since $A \neq 0$, $d$ must be nonzero.

  \Leftarrowdirection Conversely, suppose $A = cd$ where $c = (c_1, \dots, c_m)^\top \in \myF^{m,1} \setminus \{0\}$ and $d = (d_1, \dots, d_n) \in \myF^{1,n} \setminus \{0\}$.
  Then the $k$-th column of $A$ is $A_{\mathsmaller{\bullet}, k} = d_k c \in \myspan\{c\}$.
  Therefore, the column space of $A$ is contained in $\myspan\{c\}$.
  Since $c \neq 0$ and $d \neq 0$, at least one column $d_k c$ is nonzero, so the column space equals $\myspan\{c\}$.
  Hence $\operatorname{rank} A = \dim \myspan\{c\} = 1$.
\end{xprf}
\end{xrcs}

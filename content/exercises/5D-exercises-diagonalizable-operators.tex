\section*{Exercises about Diagonalizable Operators}

\exercise{2}
\begin{xrcs}
  Suppose $T \in \op(V)$ has a diagonal matrix $A=\m{T}$ with respect to some basis of $V$. Prove that if $\lambda \in F$, then $\lambda$ appears on the diagonal of $A$ precisely $\dim E (\lambda , T)$ times.

  \begin{xprf}
    If $\lambda$ is not an eigenvalue, then $\dim E(\lambda, T) = \dim \set{ v : Tv = \lambda v } = \dim \set{0} = 0$, and it does not appear on the diagonal of $A$. Now, assume $\lambda$ is an eigenvalue of $T$. Since $T$ has a diagonal matrix with respect to some basis of $V$, say $v_1, \ddd v_\ell$ with $\ell = \dim V$, all those vectors are eigenvectors of $T$ (see \ref{thm: conditions equivalent to diagonalizability}). A subcollection that satisfy $Tv = \lambda v$ form a basis of $E(\lambda, T)$. This is because every linearly independent list of vectors of the same size as the dimension of the vector space forms a basis of said vector space. Let $v_{n_1}, \ddd, v_{n_m}$ be a basis of $E(\lambda, T)$ with $\dim E(\lambda, T) = m$ and $n_j \in \set{1, \ddd, \ell}$ for $j = 1, \ddd, m$.  This implies that $T v_{n_j} = \lambda v_{n_j}$ for $j \in \set{1, \ddd, m }$, which can only happen if $\lambda$ appears on $A$ exactly $m = \dim E(\lambda, T)$ times.
  \end{xprf}
\end{xrcs}

\exercise{3}
\begin{xrcs}
  Suppose $T \in \op(V)$ has a diagonal matrix $A=\m{T}$ with respect to some basis of $V$. Prove that if $\lambda \in F$, then $\lambda$ appears on the diagonal of $A$ precisely $\dim E (\lambda , T)$ times.

  \begin{xprf}
    If $\lambda$ is not an eigenvalue, then
    \[
    \dim E(\lambda, T)
    =\dim\{v:Tv=\lambda v\}
    =\dim\{0\}=0,
    \]
    and hence $\lambda$ does not appear on the diagonal of $A$.

    Now assume $\lambda$ is an eigenvalue. Let $\mathcal B=\{v_1,\dots,v_\ell\}$ be a basis of $V$ with respect to which $A$ is diagonal, so
    $A=\operatorname{diag}(a_1,\dots,a_\ell)$ and $T v_j = a_j v_j$ for each $j$. In particular each $v_j$ is an eigenvector of $T$.

    Define \(S=\{v_j : T v_j=\lambda v_j\}\). Clearly \(\operatorname{span}S\subseteq E(\lambda,T)\). Conversely, if \(w\in E(\lambda,T)\) then writing
    \(w=\sum_{j=1}^\ell c_j v_j\) and applying \(T\) yields
    \[
    \lambda\sum_{j=1}^\ell c_j v_j
    =T w
    =\sum_{j=1}^\ell c_j T v_j
    =\sum_{j=1}^\ell c_j a_j v_j.
    \]
    By uniqueness of coordinates relative to the basis \(\mathcal B\), we must have \(c_j=0\) whenever \(a_j\neq\lambda\). Hence \(w\in\operatorname{span}S\). Therefore \(\operatorname{span}S=E(\lambda,T)\).

    Since \(S\) is a subset of the basis \(\mathcal B\), the vectors in \(S\) are linearly independent; hence \(|S|=\dim E(\lambda,T)\). But \(|S|\) is exactly the number of diagonal entries equal to \(\lambda\). Therefore \(\lambda\) appears on the diagonal of \(A\) precisely \(\dim E(\lambda,T)\) times.

    If \(w\in E(\lambda,T)\) and \(w=\sum_j c_j v_j\), then applying \(T\) gives
    \(\lambda\sum_j c_j v_j = T w = \sum_j c_j a_j v_j\), so \((\lambda-a_j)c_j=0\) for every \(j\); hence \(c_j=0\) whenever \(a_j\neq\lambda\), and thus \(w\) is a linear combination of the basis vectors with diagonal entry \(\lambda\), i.e.\ \(w\in\operatorname{span}S\).

      \end{xprf}
\end{xrcs}

\begin{xprf}
  If $\lambda^*$ is not an eigenvalue of $T$, then
  \[
  \dim E(\lambda^*,T)=\dim\{v\in V:Tv=\lambda^* v\}=\dim\{0\}=0,
  \]
  and therefore $\lambda^*$ does not appear among the diagonal entries of $A$.

  Now assume $\lambda^*$ is an eigenvalue. Let $\mathcal B=\{v_1,\dots,v_\ell\}$ be a basis of $V$ with respect to which $A=\m{T}$ is diagonal; write
  \[
  A=\operatorname{diag}(\mu_1,\dots,\mu_\ell),
  \]
  so that for each $j$ we have $T v_j=\mu_j v_j$. In particular every $v_j$ is an eigenvector of $T$ (possibly for different eigenvalues).

  Define the subset of basis vectors corresponding to the eigenvalue $\lambda^*$ by
  \[
  S=\{\, v_j \in \mathcal B \mid \mu_j=\lambda^* \,\}.
  \]
  We first observe that $\operatorname{span}S\subseteq E(\lambda^*,T)$, since each vector in $S$ is by definition a $\lambda^*$-eigenvector and linear combinations of $\lambda^*$-eigenvectors remain in the $\lambda^*$-eigenspace.

  For the reverse inclusion, let $w\in E(\lambda^*,T)$. Express $w$ in the basis $\mathcal B$:
  \[
  w=\sum_{j=1}^\ell c_j v_j \qquad (c_j\in F).
  \]
  Applying $T$ to both sides and using $T v_j=\mu_j v_j$ gives
  \[
  \lambda^* \sum_{j=1}^\ell c_j v_j = T w = \sum_{j=1}^\ell c_j \mu_j v_j.
  \]
  By uniqueness of coordinates with respect to the basis $\mathcal B$, the coefficient of $v_j$ on the left must equal the coefficient of $v_j$ on the right for each $j$, hence
  \[
  (\lambda^*-\mu_j)c_j = 0 \quad\text{for every } j=1,\dots,\ell.
  \]
  Consequently $c_j=0$ whenever $\mu_j\neq\lambda^*$, so every nonzero coefficient $c_j$ corresponds to some $v_j\in S$. Thus $w$ is a linear combination of vectors from $S$, meaning $w\in\operatorname{span}S$. This proves the reverse inclusion and therefore
  \[
  \operatorname{span}S = E(\lambda^*,T).
  \]

  Since $S$ is a subset of the basis $\mathcal B$, its vectors are linearly independent; hence the number of elements of $S$ equals the dimension of its span. Therefore
  \[
  |S| = \dim\operatorname{span}S = \dim E(\lambda^*,T).
  \]
  Finally, by definition $|S|$ is exactly the number of diagonal entries of $A$ equal to $\lambda^*$. We conclude that the scalar $\lambda^*$ appears on the diagonal of $A$ precisely $\dim E(\lambda^*,T)$ times.
\end{xprf}

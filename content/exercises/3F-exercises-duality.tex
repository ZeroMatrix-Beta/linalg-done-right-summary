\section*{Exercises about Duality}

\exercise{1}

\begin{xrcs}
  Explain why each linear functional is surjective or the zero map.

  \begin{xsol}
    $\varphi \in \dual{V} = \lin{V}{\myF}$ implies that for every $\lambda \in \myF$, and every $v \in V$,
    \begin{equation}
      \varphi (\lambda v) = \lambda \varphi(v).
    \end{equation}

    If $\varphi$ is not the zero map, then $\varphi$ evaluated at $v \in V$ or $\lambda v \in V$ is a scalar in  $\myF$ for some $v \in V$ for which it holds that $\varphi(v) \neq 0$. In this case, we have $\varphi(v) \in \myF$ and $ \varphi(\lambda v) = \lambda \varphi(v) \in \myF$. Hence, $\myrange \varphi = \myF$, since $\lambda  \in \myF$ was arbitrary.
  \end{xsol}
\end{xrcs}

\exercise{3}
\begin{xrcs}
  Suppose $V$ is finite-dimensional and $v \in V$ is nonzero. Prove that there exists $\varphi \in \dual{V}$ such that $\varphi(v) = 1$.

  \begin{xsol}
    Let $v_1, \ddd, v_n$ be a basis of $V$. Let $v \in V$ be nonzero such that $v = a_1 v_1 + \cdots + a_n v_n$, where $a_i \in \myF$. Since $v$ is nonzero, there exists a $k$ such that $a_k \neq 0$. Define $\varphi: V \to \real$ by
    \begin{equation}
      \varphi (u) = \frac{c_k}{a_k},
    \end{equation}

    whenever $u = c_1 v_1 + \dots + c_n v_n \in V$. Note that this definition corresponds to the use of the linear map lemma (\ref{thm: linear map lemma}), where we define $\varphi$ on each basis vector as follows:
    \begin{equation}
      \varphi (v_1) = 0, \;  \ddd, \;
      \varphi(v_{k-1}) = 0, \;
      \varphi(v_k) = \frac{1}{a_k}, \;
      \varphi(v_{k+1}) = 0, \;  \ddd, \;
      \varphi(v_n) = 0.
    \end{equation}

    Hence, $\varphi(a_1 v_1 + \cdots + a_n v_n) = \frac{a_k}{a_k} = 1$. It is easy to check that $\varphi$ is a linear map.
  \end{xsol}
\end{xrcs}

\exercise{5}
\begin{xrcs}
  Suppose $V$ is finite dimensional and $U$ is a subspace to $V$, where $U \neq V$. Prove that there exists a $\varphi \in \dual{V}$ such that $\varphi (u) = 0$ for every $u \in U$ but $\varphi \neq 0$.

  \begin{xsol}
    Choose $W \subset V$ such that $V = U \oplus W$. Hence, there is a basis
    \begin{equation}
      u_1, \ddd, u_n, w_1, \ddd, w_m
    \end{equation}
    of $V$ such that the $u$'s and $w$'s form a basis of $U$ and $W$ respectively. We define $\varphi \in \dual{V}$ such that for $v \in V$, written as
    \begin{equation}
      v = a_1 u_1 + \cdots + a_n u_n + b_1 w_1 + \cdots + b_m w_m,
    \end{equation}

    we have
    \begin{equation}
      \varphi (v) = b_1 + \cdots + b_m.
    \end{equation}

    It is easy to see that $\varphi (u) = 0$ for every $u \in U$ (since $U \cap W = \{0\}$, by \ref{thm: sum and intersection of two subspaces}) and that $\varphi$ is linear.
  \end{xsol}
\end{xrcs}

\exercise{6}
\begin{xrcs}
  Suppose $\varphi, \beta \in \dual{V}$. Prove that $\mynull \varphi \subseteq \mynull \beta$ if and only if there exists $c \in \myF$ such that
  \begin{equation}
    \beta = c \varphi.
  \end{equation}

\begin{xprf}
  \Leftarrowdirection If $\beta = c \varphi$, then $\varphi(v) = 0 \implies \beta(v) = c \varphi(v) = 0$, so $\mynull \varphi \subseteq \mynull \beta$.

  \Rightarrowdirection Conversely, suppose $\mynull \varphi \subseteq \mynull \beta$.
  If $\varphi = 0$, then $\mynull \varphi = V \subseteq \mynull \beta \implies \beta = 0 = 0\varphi$, so take $c = 0$.
  If $\varphi \neq 0$, choose $w \in V$ such that $\varphi(w) \neq 0$.
  For any $v \in V$, consider $u = v - \frac{\varphi(v)}{\varphi(w)} w$.
  Then $\varphi(u) = \varphi(v) - \frac{\varphi(v)}{\varphi(w)} \varphi(w) = 0$, so $u \in \mynull \varphi \subseteq \mynull \beta$.
  Thus $\beta(u) = 0$, which gives $\beta(v) - \frac{\varphi(v)}{\varphi(w)} \beta(w) = 0 \implies \beta(v) = \frac{\beta(w)}{\varphi(w)} \varphi(v)$.
  Setting $c = \frac{\beta(w)}{\varphi(w)} \in \myF$ yields $\beta = c\varphi$.
\end{xprf}
\end{xrcs}

\exercise{10}
\begin{xrcs}
  Suppose $m$ is a positive integer.
  \begin{enumerate}
    \item Show that $1, x-5, \ddd, (x-5)^m$ is a basis of $\polyn_m(\real)$.
    \item What is the dual basis of the basis in (a)?
  \end{enumerate}

  \begin{xsol}
    We first start the exercise by solving it for $\polyn_4(\real)$ before solving it abstractly for $\polyn_m(\real)$. Recall that $e_0 = x^0, e_1 = x^1, \ddd, e_4 = x^4$ is the standard basis of $\polyn_4(\real)$. We want to show that $v_0 = 1, v_1 = x-5, \ddd, v_4 = (x-5)^4$ is a basis as well. Expanding the polynomials:
    \begin{equation}
      \begin{aligned}
        v_0 = (x-5)^0 &= 1x^0, \\
        v_1 = (x-5)^1 &= -5x^0 + x^1, \\
        v_2 = (x-5)^2 &= 25x^0 - 10x^1 + x^2, \\
        v_3 = (x-5)^3 &= -125x^0 + 75x^1 - 15x^2 + x^3, \\
        v_4 = (x-5)^4 &= 625x^0 - 500x^1 + 150x^2 - 20x^3 + x^4.
      \end{aligned}
    \end{equation}

    Now, for a given $p \in \polyn_4(\real)$ with $p(x) = b_0 x^0 + b_1 x^1 + \cdots + b_4 x^4$ ($b_j \in \real$), we show that there exist $a_j \in \real$ such that $\sum_{j=0}^{4} b_j e_j = \sum_{j=0}^{4} a_j v_j$. Expanding $\sum_{j=0}^4 a_j v_j$ and matching coefficients with $\sum_{j=0}^4 b_j x^j$:
    \begin{equation}
      \begin{aligned}
        a_4 &= b_4, \\
        a_3 &= b_3 + 20 a_4, \\
        a_2 &= b_2 + 15 a_3 - 150 a_4, \\
        a_1 &= b_1 + 10 a_2 - 75 a_3 + 500 a_4, \\
        a_0 &= b_0 + 5a_1 - 25 a_2 + 125 a_3 - 625 a_4.
      \end{aligned}
    \end{equation}

    Since this triangular system can always be solved uniquely by back-substitution, $v_0, \ddd, v_4$ is indeed a basis of $\polyn_4(\real)$.

    Now, we find the dual basis $\psi_0, \psi_1, \ddd, \psi_4$ of $v_0, \ddd, v_4$. Recall the duality condition:
    \begin{equation}
      \psi_j (v_k) = \delta_{j,k} = \begin{cases}
        1, & j=k, \\
        0, & j \neq k.
      \end{cases}
    \end{equation}

    More generally, using the binomial theorem on $(x-5)^k = \sum_{i=0}^k \binom{k}{i} (-5)^{k-i} x^i$, we observe that the linear functional $\psi_j$ defined by
    \begin{equation}
      \psi_j(p) := \frac{1}{j!} p^{(j)}(5)
    \end{equation}
    satisfies $\psi_j((x-5)^k) = \frac{1}{j!} \frac{d^j}{dx^j}(x-5)^k\big|_{x=5} = \delta_{j,k}$.
    Thus, the dual basis is given by $\psi_j(p) = \frac{p^{(j)}(5)}{j!}$ for $j = 0, \dots, m$.
  \end{xsol}
\end{xrcs}

\exercise{11}
\begin{xrcs}
  Suppose $v_1, \ddd, v_n$ is a basis of $V$ and $\varphi_1, \ddd, \varphi_n$ is the corresponding dual basis of $\dual{V}$. Suppose $\psi \in \dual{V}$. Prove that
  \begin{equation}
     \label{eq: equation for psi}
     \psi = \psi(v_1) \varphi_1 + \cdots +  \psi(v_n) \varphi_n.
  \end{equation}

  \begin{xsol}
    For a given $\psi \in \dual{V}$, define $\Gamma \in \dual{V}$ by
    \begin{equation}
      \Gamma := \psi(v_1) \varphi_1 + \cdots +  \psi(v_n) \varphi_n.
    \end{equation}
    We want to show that $\Gamma = \psi$. Applying $\Gamma$ to each basis vector $v_k$ ($k = 1, \ddd, n$):
    \begin{equation}
      \begin{aligned}
        \Gamma(v_k) &= (\psi(v_1) \varphi_1 + \cdots +  \psi(v_n) \varphi_n) (v_k) \\
                    &= \psi(v_1) \varphi_1 (v_k) + \cdots +  \psi(v_k) \varphi_k (v_k) + \cdots +  \psi(v_n) \varphi_n (v_k) \\
                    &= \psi(v_1) \cdot 0 + \cdots +  \psi(v_k) \cdot 1 + \cdots +  \psi(v_n) \cdot 0 \\
                    &= \psi(v_k).
      \end{aligned}
    \end{equation}
    Hence, $\Gamma$ and $\psi$ agree on every vector of the basis $v_1, \dots, v_n$. Since a linear map is uniquely determined by its values on a basis, $\Gamma = \psi$.
  \end{xsol}
\end{xrcs}

\exercise{14}
\begin{xrcs}
  Define $T: \real^3 \to \real^2$ by
  \begin{equation}
    T(x, y, z) := (4x + 5y + 6z, 7x + 8y + 9z).
  \end{equation}

  Suppose $\varphi_1, \varphi_2$ denotes the dual basis of the standard basis of $\real^2$ and $\psi_1, \psi_2, \psi_3$ denotes the dual basis of the standard basis of $\real^3$.
  \begin{enumerate}
    \item Describe the linear functionals $T'(\varphi_1)$ and $T'(\varphi_2)$.
    \item Write $T'(\varphi_1)$ and $T'(\varphi_2)$ as a linear combination of $\psi_1, \psi_2, \psi_3$.
  \end{enumerate}

  \begin{xsol}
    Let's start by getting comfortable about the notation. Let $e_1, e_2, e_3$ denote the standard basis of $\real^3$ and $e_1, e_2$ denote the standard basis of $\real^2$. In this case, $T(x, y, z) = (4x + 5y + 6z, 7x + 8y + 9z)$ actually means
    \begin{equation}
      T(x e_1 + y e_2 + z e_3) = (4x + 5y + 6z) e_1 + (7x + 8y + 9z) e_2
    \end{equation}

    Thus, we have $T e_1 = 4e_1 + 7 e_2$, $T e_2 = 5e_1 + 8e_2$, and $T e_3 = 6e_1 + 9 e_2$. The Matrix $\mmatrix(T)$ of $T$ would look like this:
    \begin{equation}
      \mathcal{M} (T) =
      \begin{blockarray}{cccc}
        & e_1 & e_2 & e_3  \\
        \begin{block}{c(ccc)}
          e_1 & 4   & 5   & 6    \\
          e_2 & 7   & 8   & 9    \\
        \end{block}
      \end{blockarray}
    \end{equation}

    Now, $\dual{T} (\varphi_1) = \varphi_1 \circ T$ and $\dual{T} (\varphi_2) = \varphi_2 \circ T$. Hence,
    \begin{equation}
      \begin{aligned}
        &\dual{T} (\varphi_1) (x e_1 + y e_2 + z e_3) = \varphi_1 ((4x + 5y + 6z) e_1 + (7x + 8y + 9z) e_2 ) = 4x + 5y + 6z \;, \; \text{ and } \\
        &\dual{T} (\varphi_2) (x e_1 + y e_2 + z e_3) = \varphi_2 ((4x + 5y + 6z) e_1 + (7x + 8y + 9z) e_2 ) = 7x + 8y + 9z,
      \end{aligned}
    \end{equation}

    which answers question (a). For question (b), we use the result \eqref{eq: equation for psi} from question 11 and write
    \begin{equation}
      \begin{aligned}
        &\dual{T} (\varphi_1) =  \sum_{k=1}^{3} \dual{T} (\varphi_1) (e_k) \psi_k = 4 \psi_1 + 5 \psi_2 + 6 \psi_3 \; , \text{ and }  \\
        &\dual{T} (\varphi_2) =  \sum_{k=1}^{3} \dual{T} (\varphi_2) (e_k) \psi_k = 7 \psi_1 + 8 \psi_2 + 9 \psi_3 \;
      \end{aligned}
    \end{equation}


    \prooffont{Variation of Qustion 14:} The question becomes more interesting if we use another basis for $\real^2$ and $\real^3$, say $v_1 = (1,1), v_2=(2,1)$ for $\real^2$ and $w_1 = (2,2,0), w_2 = (4,2,0), w_3 = (0,0,1)$ for $\real^3$. In this case, let $\phi_1, \phi_2$ denote the dual basis of $v_1, v_2$ and $\psi_1, \psi_2, \psi_3$ denote the dual basis of $w_1, w_2, w_3$. Now, we have to calculate $\phi$ and $\psi$ first. Per definition, we have
    \begin{equation}
      \phi(v) = \phi (a_1 v_1 + a_2 v_2) = a_1 \phi_1 (v_1) + a_2 \phi_2 (v_2).
    \end{equation}

    And we must assure that $\phi_1(v_1)=1, \phi(v_2)=0$ and $\phi_2(v_2)=0$ and $\phi_2(v_2) = 1$. Using the facts that $v_1 = e_1 + e_2$ and $v_2 = 2 e_1 + e_2$, we get
    \begin{equation}
      \left | \;
      \begin{aligned}
        1 \phi_1 (e_1) + 1 \phi_1 (e_2) &= 1 \\
        2 \phi_1 (e_1) + 1 \phi_1 (e_2) &= 0
      \end{aligned} \;
      \right  | \; \implies \;
      \left | \;
        \begin{aligned}
          1 \phi_1 (e_1) + 1 \phi_1 (e_2) &= 1 \\
                         - 2 \phi_1 (e_2) &= -2
        \end{aligned} \;
      \right | \; ,
    \end{equation}

    and hence, $\phi_1(e_1) = -1$ and $\phi_1(e_2) = 2$. Likewise,
    \begin{equation}
    \left | \;
    \begin{aligned}
      1 \phi_2 (e_1) + 1 \phi_2(e_2) &= 0 \\
      2 \phi_2 (e_1) + 1 \phi_2 (e_2) &= 1
    \end{aligned} \;
    \right  | \; \implies \;
    \left | \;
    \begin{aligned}
      1 \phi_2 (e_1) + 1 \phi_2 (e_2) &= 0 \\
      - 2 \phi_2 (e_2) &= 1
    \end{aligned} \;
    \right | \; .
    \end{equation}

    Therefore, $\phi_2(e_1) = 1$ and $\phi_2(e_2) = -1$. To write $\phi_1$ and $\phi_2$ as members of $\real^2 \to \real$, we have $\phi_1(x,y) = -x + 2y$ and $\phi_2 (x,y) = x-y$. Now,
    \begin{equation}
      \begin{aligned}
        T'(\phi_1) &= \phi_1 \circ T = -(4x + 5y + 6z) + 2          (7x + 8y + 9z) = 10x + 11y + 12z \; , \;  \text{ and }  \\
        T'(\phi_2) &= \phi_2 \circ T = 4x + 5y + 6z - (7x + 8y + 9z) = -3x -3y -3z = -3(x+y+z).
      \end{aligned}
    \end{equation}

    With the same approach, we can calculate the dual basis for $w_1, w_2, w_3$. We get $\psi_1(x,y,z) = -\frac{1}{2}x + y$, $\psi_2(x,y,z) = \frac{1}{2}x - \frac{1}{2}y$ and $\psi_3(x,y,z) = z$. To answer question (b), we use the theorem \eqref{eq: equation for psi} from question 11 and write
    \begin{equation}
      \begin{aligned}
        \dual{T} (\varphi_1) =  \sum_{k=1}^{3} \dual{T} (\varphi_1) (w_k) \psi_k &= (20+22+0) \psi_1 + (40+22+0) \psi_2 + (0+0+12 )\psi_3 \\
        & = 42 \psi_1 + 62 \psi_2 + 12 \psi_3  \; , \text{ and }  \\
        \dual{T} (\varphi_2) =  \sum_{k=1}^{3} \dual{T} (\varphi_2) (w_k) \psi_k &= -12 \psi_1 -18 \psi_2 -3 \psi_3. \;
      \end{aligned}
    \end{equation}
  \end{xsol}
\end{xrcs}

\exercise{15}
\begin{xrcs}
  Suppose $W$ is finite-dimensional and $T \in \linmap (V,W)$. Prove that
  \begin{equation}
    \dual{T} = 0 \iff T = 0.
  \end{equation}
  \begin{xprf}
    \Rightarrowdirection Assume $\dual{T} = 0$. By definition, $\dual{T}(\varphi) = \varphi \circ T = 0$ for every $\varphi \in \dual{W}$. If $W = \{0\}$, then necessarily $T = 0$, so $\dual{T} = 0$ holds trivially.
    For this reason, assume for now that $W \neq 0$ and let $w_1, \ddd, w_n$ be a basis of $W$. Suppose for the sake of contradiction that $T \neq 0$. Hence, there exists a nonzero $v \in V$ and a nonzero $w \in W$ such that $w=T(v)$. Express $w$ as $w=a_1 w_1 + \cdots + a_n w_n$. Since $w \neq 0$, pick $k$ such that $a_k \neq 0$. Define $\varphi \in \dual{W}$ by
    \begin{equation}
      \varphi(b_1 w_1 + \cdots + b_n w_n) = \frac{b_k}{a_k},
    \end{equation}

    where $b_k$ is the $k$-th coordinate of $\varphi$'s input. Hence, $\varphi(w) =1$.
    But then
    \begin{equation}
      \dual{T}(\varphi) (v) = \varphi(T(v)) = \varphi(w) = 1 \neq 0.
    \end{equation}

    Thus, $\dual{T}(\varphi) \neq 0$, contradicting $\dual{T} = 0$. Hence, $T = 0$.

    \Leftarrowdirection If $T = 0$, then for each $\varphi \in \dual{W}$ and $\dual{T} \in \lin{\dual{W}}{\dual{V}}$ we have
    \begin{equation}
      \dual{T} (\varphi) (v) = \varphi(T(v)) = \varphi(0) = 0.
    \end{equation}

    Hence, $\dual{T} (\varphi)$ is always sent to a map in $\dual{W}$ that is the zero map.
  \end{xprf}
\end{xrcs}


\exercise{16}
\begin{xrcs}
  Suppose $V$ and $W$ are finite-dimensional and $T \in \linmap(V,W)$. Prove that
  \begin{equation}
    T \text{ is invertible} \iff \dual{T} \in \linmap(\dual{W}, \dual{V}) \text{ is invertible.}
  \end{equation}

  \begin{xprf}
    \Rightarrowdirection Let $T \in \linmap(V,W)$ be invertible and let $T^{-1} \in \linmap(W,V)$ be the inverse of $T$. We first want to show that $\dual{\left(T^{-1} \right)}$ is the inverse of $\dual{T}$, that is,
    \begin{equation}
      \dual{\left(T^{-1} \right)} = \left(\dual{T}\right)^{-1}.
    \end{equation}

    For $\dual{\left(T^{-1}\right)} \in \lin{\dual{V}}{\dual{W}}$ to be the inverse of $\dual{T} \in \lin{\dual{W}}{\dual{V}}$, it must hold that $\dual{\left(T^{-1}\right)} \dual{T} = I_{\dual{W}}$ and $\dual{T} \dual{\left(T^{-1}\right)} = I_{\dual{V}}$. For any $\psi \in \dual{W}$, we compute:
    \begin{equation}
      \left(\dual{\left(T^{-1}\right)} \circ \dual{T}\right)(\psi) = \dual{\left(T^{-1}\right)}(\psi \circ T)
      = (\psi \circ T) \circ T^{-1} = \psi \circ (T T^{-1}) = \psi \circ I_W = \psi.
    \end{equation}

    Thus, $\dual{\left(T^{-1}\right)} \dual{T} = I_{\dual{W}}$. For the other direction, let $\varphi \in \dual{V}$:
    \begin{equation}
      \left(\dual{T} \circ \dual{\left(T^{-1}\right)}\right)(\varphi) = \dual{T}(\varphi \circ T^{-1})
      = (\varphi \circ T^{-1}) \circ T = \varphi \circ (T^{-1} T) = \varphi \circ I_V = \varphi,
    \end{equation}

    so $\dual{T} \dual{\left(T^{-1}\right)} = I_{\dual{V}}$. Hence, $\dual{T}$ is invertible with $\left(\dual{T}\right)^{-1} = \dual{\left(T^{-1}\right)}$.

    \Leftarrowdirection If $\dual{T}$ is invertible, then it is surjective (and also injective) by \ref{thm: injectivity is equivalent to surjectivity}. Take any nonzero $v \in V$ and $\varphi \in \dual{V}$ with $\varphi (v) \neq 0$. Surjectivity of $\dual{T}$ yields a $\psi \in \dual{W}$ with $\dual{T}(\psi) = \varphi$, meaning $\varphi = \psi \circ T$. Therefore,
    \begin{equation}
      0 \neq \varphi(v) = \psi(Tv) = (\psi \circ T)(v).
    \end{equation}

    This implies that $Tv \neq 0$, and thus $\mynull T = \{0\}$ (since $v$ was arbitrary and nonzero). This forces $T$ to be injective. Because $\dual{T}$ is invertible, we have $\dim \dual{W} = \dim \dual{V}$, and hence $\dim W = \dim V$ (see \ref{thm: dim V' = dim V}). An injective linear map between finite-dimensional vector spaces of the same dimension is surjective, so $T$ is bijective, and therefore, invertible.
  \end{xprf}
\end{xrcs}

\exercise{19}
\begin{xrcs}
  Suppose $U$ and $W$ are subspaces of a finite-dimensional vector space $V$. Prove that
  \begin{equation}
    (U + W)^0 = U^0 \cap W^0 \quad \text{and} \quad (U \cap W)^0 = U^0 + W^0.
  \end{equation}
\begin{xprf}
  \StepOne We prove $(U+W)^0 = U^0 \cap W^0$.
  Let $\varphi \in \dual{V}$. Then $\varphi \in (U+W)^0 \iff \varphi(u+w) = 0$ for all $u \in U, w \in W$.
  Taking $w = 0$ gives $\varphi(u) = 0$ for all $u \in U$ ($\varphi \in U^0$), and taking $u = 0$ gives $\varphi(w) = 0$ for all $w \in W$ ($\varphi \in W^0$).
  Thus $\varphi \in (U+W)^0 \iff \varphi \in U^0 \cap W^0$.

  \StepTwo We prove $(U \cap W)^0 = U^0 + W^0$.
  Since $U \cap W \subseteq U$ and $U \cap W \subseteq W$, taking annihilators gives $U^0 \subseteq (U \cap W)^0$ and $W^0 \subseteq (U \cap W)^0$, so $U^0 + W^0 \subseteq (U \cap W)^0$.
  To show equality, we compare dimensions:
  \[
  \begin{aligned}
    \dim(U^0 + W^0) &= \dim U^0 + \dim W^0 - \dim(U^0 \cap W^0) \\
    &= (\dim V - \dim U) + (\dim V - \dim W) - \dim((U+W)^0) \\
    &= 2\dim V - \dim U - \dim W - (\dim V - \dim(U+W)) \\
    &= \dim V - (\dim U + \dim W - \dim(U+W)) = \dim V - \dim(U \cap W) = \dim((U \cap W)^0).
  \end{aligned}
  \]
  Since $U^0 + W^0$ is a subspace of $(U \cap W)^0$ of the same dimension, $U^0 + W^0 = (U \cap W)^0$.
\end{xprf}
\end{xrcs}
